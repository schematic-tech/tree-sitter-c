#ifdef PLATFORM_A
    if (read_mem(ptr, pid, start, len)) {
#else
    if (read_mem(ptr, fd, start, len)) {
#endif
        report_error();
    }
#if VERSION == 1
    if (old_read(ptr)) {
#elif VERSION == 2
    if (current_read(ptr)) {
#else
    if (new_read(ptr)) {
#endif
        retry();
    } else {
        finish();
    }
}
